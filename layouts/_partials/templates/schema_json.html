{{- /* Structured data (JSON-LD) for every page. Overrides PaperMod's templates/schema_json.html so the author is a real, linked Person entity instead of a bare name. "@type" "Person" "@id" $personID "name" (site.Params.author | default site.Title) "url" $aboutURL "sameAs" $sameAs -}} {{- with site.Params.schema.person -}} {{- with .jobTitle }}{{ $person = merge $person (dict "jobTitle" .) }}{{ end -}} {{- with .description }}{{ $person = merge $person (dict "description" .) }}{{ end -}} {{- with .knowsAbout }}{{ $person = merge $person (dict "knowsAbout" .) }}{{ end -}} {{- with .image }}{{ $person = merge $person (dict "image" (absURL .)) }}{{ end -}} {{- end -}} {{- /* Short reference form — enough for author/publisher without repeating the bio. */ -}} {{- $personRef := dict "@type" "Person" "@id" $personID "name" $person.name "url" $aboutURL -}} {{- /* Page image: explicit cover first, then whatever PaperMod would pick for og:image. */ -}} {{- $img end -}} {{- else -}} {{- with partial "templates/_funcs/get-page-images" . }}{{ with index . 0 }}{{ $img = .Permalink }}{{ end }}{{ end -}} {{- end -}} {{- /* Dates — omitted when Hugo has nothing real (the About page has no date). */ -}} {{- $pub := "" }}{{ if not .PublishDate.IsZero }}{{ $pub = .PublishDate.Format "2006-01-02T15:04:05Z07:00" }}{{ end -}} {{- $mod := "" }}{{ if not .Lastmod.IsZero }}{{ $mod = .Lastmod.Format "2006-01-02T15:04:05Z07:00" }}{{ end -}} {{- if .IsHome -}} {{- $website := dict "@type" "WebSite" "@id" $websiteID "name" site.Title "url" -}} {{- range $index, $element := $bc_list }} {{- $scratch.Add "path" (printf "%s/" $element ) }} {{- $bc_pg := site.GetPage ($scratch.Get "path") -}} {{- if (and ($bc_pg) (gt (len . ) 0)) }} {{-
